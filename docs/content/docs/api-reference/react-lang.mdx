## `defineComponent(config)`

Defines a single component with name, Zod schema, description, and React renderer. Returns a `DefinedComponent` with a `.ref` for cross-referencing in parent schemas.

```ts
function defineComponent<T extends z.ZodObject<any>>(config: {
  name: string;
  props: T;
  description: string;
  component: ComponentRenderer<z.infer<T>>;
}): DefinedComponent<T>;
```

```ts
interface DefinedComponent<T extends z.ZodObject<any> = z.ZodObject<any>> {
  name: string;
  props: T;
  description: string;
  component: ComponentRenderer<z.infer<T>>;
  /** Use in parent schemas: `z.array(ChildComponent.ref)` */
  ref: z.ZodType<SubComponentOf<z.infer<T>>>;
}
```

## `createLibrary(input)`

Creates a `Library` from an array of defined components.

```ts
/** The result has functions, actions, and extend set (Required<Library>). */
function createLibrary(input: LibraryDefinition): Required<Library>;
```

Core types:

```ts
interface LibraryDefinition {
  components: DefinedComponent[];
  componentGroups?: ComponentGroup[];
  /** Custom functions, called as `@Name(...)` in programs. */
  functions?: DefinedFunction[];
  /** Custom action steps, used as `@Name(...)` in action props. */
  actions?: DefinedAction[];
  root?: string;
  /** Optional public id. `toSpec()` includes it. */
  id?: string;
}

interface ComponentGroup {
  name: string;
  components: string[];
  notes?: string[];
}

interface Library {
  readonly components: Record<string, DefinedComponent>;
  readonly componentGroups: ComponentGroup[] | undefined;
  /** Optional, so a hand-built Library object still type-checks. createLibrary always sets it. */
  readonly functions?: Record<string, DefinedFunction>;
  /** Optional, like functions. */
  readonly actions?: Record<string, DefinedAction>;
  readonly root: string | undefined;
  readonly id: string | undefined;

  prompt(options?: PromptOptions): string;
  toJSONSchema(): LibraryJSONSchema;
  toSpec(): PromptSpec;
  /** Derive a new library. Optional, like functions. */
  extend?(extension: LibraryExtension): Library;
}

interface PromptOptions {
  preamble?: string;
  additionalRules?: string[];
  examples?: string[];
  toolExamples?: string[];
  editMode?: boolean;
  inlineMode?: boolean;
  /** Enable Query(), Mutation(), @Run, built-in functions. Default: true if tools provided. */
  toolCalls?: boolean;
  /** Enable $variables, @Set, @Reset, built-in functions. Default: true if toolCalls. */
  bindings?: boolean;
  /** List the built-in functions (@Count, @Sum, ...). Default: true if toolCalls or bindings. */
  builtinFunctions?: boolean;
}
```

`createLibrary` throws when a function or action name is a built-in name, `Action`, `Query`, `Mutation`, or the name of a component, function, or action in the same library.

## `defineFunction(config)`

Defines a custom function, called as `@Name(...)` in programs. See [Custom Functions](/docs/openui-lang/custom-functions).

```ts
function defineFunction<
  T extends z.ZodObject,
  R extends z.ZodType | undefined = undefined,
>(config: {
  name: string;
  description: string;
  /** Key order is the positional order of the arguments. */
  params: T;
  /** Printed in the prompt signature and checked at runtime. */
  returns?: R;
  /** Receives one object keyed by the params names, with defaults applied. */
  fn: (args: z.infer<T>) => R extends z.ZodType ? z.infer<R> : unknown;
}): DefinedFunction<T, R>;
```

A call with an invalid required argument, an `fn` that throws, or a result that does not match `returns` evaluates to `null`. An invalid optional argument falls back to its default.

## `defineAction(config)`

Defines a custom action step. The renderer does not run it: it calls `onAction` with `{ type: name, params }`. A step with any invalid argument is a no-op. See [Custom Actions](/docs/openui-lang/custom-actions).

```ts
function defineAction<const N extends string, T extends z.ZodObject>(config: {
  name: N;
  description: string;
  /** Key order is the positional order of the arguments. */
  params: T;
}): DefinedAction<T, N>;

interface DefinedAction<T, N> {
  name: N;
  description: string;
  params: T;
  /** For slots that take only some actions: `z.union([copy.ref, steps.OpenUrl.ref])` */
  ref: ActionRef;
}
```

## `action()` and `steps`

`action()` returns the schema for an action prop, tagged `ActionExpression`. In the JSON Schema the prop is `{"$ref": "#/$defs/ActionExpression"}`.

```ts
props: z.object({ label: z.string(), action: action().optional() });
```

`steps.OpenUrl.ref` and `steps.ToAssistant.ref` name built-in steps in a restricted slot, together with the `.ref` of custom actions. Restricted slots shape the prompt and the JSON Schema only; the runtime accepts any step. See [Action Expressions](/docs/openui-lang/action-expressions).

## `library.extend(extension)`

Returns a new library with the additions. The base library is unchanged. See [Extending a Library](/docs/openui-lang/extending-libraries).

```ts
interface LibraryExtension {
  components?: {
    /** With slots ("Parent.prop", or "Parent" when it has one content slot), it joins those unions. */
    add?: (DefinedComponent | { component: DefinedComponent; slots: string[] })[];
    /** Replaces the component of the same name. Every parent points at the new one. */
    override?: DefinedComponent[];
    /** Drops it from the library, every parent union, and componentGroups. */
    remove?: string[];
  };
  functions?: { add?: DefinedFunction[] };
  actions?: { add?: DefinedAction[] };
}
```

`extend` throws when an added name exists, an overridden or removed name does not, a removal is the root or the only type of a slot, or a slot does not resolve to one content slot. The derived library has no `id`. Create it once at module scope, since parsers are cached per library object.

## `buildMessage(input)` and `parseMessage(raw, options?)`

Write and read a stored message with `]]>openui:` marker lines. See [Storing Messages](/docs/openui-lang/messages).

```ts
function buildMessage(input: {
  content: string;
  /** A string is written as is, anything else as JSON. Omitted when null or undefined. */
  context?: unknown;
  attributes?: Record<string, string>;
  end?: boolean;
}): string;

function parseMessage(
  raw: string,
  options?: {
    /** Hold back a last line that may be a marker cut mid-stream. */ streaming?: boolean;
  },
): {
  content: string;
  /** Parsed JSON, else raw text. null when absent. */
  context: unknown;
  attributes: Record<string, string>;
  end: boolean;
};
```

## `<Renderer />`

Parses OpenUI Lang text and renders nodes with your `Library`.

```ts
interface RendererProps<L extends Library = Library> {
  /** OpenUI Lang text, or a stored message with ]]>openui: marker lines. */
  response: string | null;
  library: L;
  isStreaming?: boolean;
  /** Typed from the library: checking event.type narrows params for custom actions. */
  onAction?: (event: LibraryActionEvent<L>) => void;
  onStateUpdate?: (state: Record<string, unknown>) => void;
  initialState?: Record<string, any>;
  onParseResult?: (result: ParseResult | null) => void;
  toolProvider?:
    | Record<string, (args: Record<string, unknown>) => Promise<unknown>>
    | McpClientLike
    | null;
  queryLoader?: React.ReactNode;
  onError?: (errors: OpenUIError[]) => void;
}
```

## Tool Provider

Handles `Query()` and `Mutation()` tool calls at runtime. The `toolProvider` prop accepts two forms:

- **Function map** — `Record<string, (args) => Promise<unknown>>` — the simplest option
- **MCP client** — any object implementing `callTool({ name, arguments })` (e.g. from `@modelcontextprotocol/sdk`)

The Renderer detects which form was passed and normalizes internally.

## Error types

```ts
type OpenUIErrorSource = "parser" | "runtime" | "query" | "mutation";

interface OpenUIError {
  source: OpenUIErrorSource;
  code: string;
  message: string;
  statementId?: string;
  component?: string;
  path?: string;
  hint?: string;
  /** "warning" when the page still rendered; absent means error. */
  severity?: "warning";
}

class ToolNotFoundError extends Error {
  toolName: string;
  availableTools: string[];
}
```

Error codes: `unknown-component`, `missing-required`, `null-required`, `type-mismatch`, `excess-args`, `inline-reserved`, `unknown-function`, `no-root`, `tool-not-found`, `tool-error`, `mcp-error`, `parse-failed`, `parse-exception`, `runtime-error`, `render-error`.

## Actions

```ts
enum BuiltinActionType {
  ContinueConversation = "continue_conversation",
  OpenUrl = "open_url",
}

interface ActionEvent {
  type: string;
  params: Record<string, any>;
  humanFriendlyMessage: string;
  formState?: Record<string, any>;
  formName?: string;
}
```

Action steps (runtime types from the evaluator):

```ts
type ActionStep =
  | { type: "run"; statementId: string; refType: "query" | "mutation" }
  | { type: "continue_conversation"; message: string; context?: unknown }
  | { type: "open_url"; url: string }
  | { type: "set"; target: string; valueAST: ASTNode }
  | { type: "reset"; targets: string[] }
  | { type: "custom_action"; name: string; params: Record<string, unknown> };
```

Every step call evaluates to a plan, `{ steps: ActionStep[] }`. A step that cannot run gives a plan with no steps.

| Step type                 | Triggered by          | Description                                                        |
| ------------------------- | --------------------- | ------------------------------------------------------------------ |
| `"run"`                   | `@Run(ref)`           | Execute a Mutation or re-fetch a Query. `refType` indicates which. |
| `"set"`                   | `@Set($var, val)`     | Change a `$variable`. `valueAST` is evaluated at click time.       |
| `"reset"`                 | `@Reset($a, $b)`      | Restore `$variables` to declared defaults.                         |
| `"continue_conversation"` | `@ToAssistant("msg")` | Send message to LLM. Optional `context` of any type.               |
| `"open_url"`              | `@OpenUrl("url")`     | Open URL in new tab.                                               |
| `"custom_action"`         | `@Name(...)`          | A custom action. Reaches `onAction` as `{ type: name, params }`.   |

## Parser APIs

Both `createParser` and `createStreamingParser` accept a `LibraryJSONSchema` (from `library.toJSONSchema()`).

```ts
interface LibraryJSONSchema {
  $defs?: Record<
    string,
    {
      properties?: Record<string, unknown>;
      required?: string[];
    }
  >;
}

function createParser(schema: LibraryJSONSchema): Parser;
function createStreamingParser(schema: LibraryJSONSchema): StreamParser;

interface Parser {
  parse(input: string): ParseResult;
}

interface StreamParser {
  push(chunk: string): ParseResult;
  getResult(): ParseResult;
}
```

Core parsed types:

```ts
interface ElementNode {
  type: "element";
  typeName: string;
  props: Record<string, unknown>;
  partial: boolean;
}

/**
 * Parser-level validation errors (schema mismatches).
 */
type ValidationErrorCode =
  | "missing-required"
  | "null-required"
  | "unknown-component"
  | "inline-reserved"
  | "excess-args"
  | "type-mismatch"
  | "unknown-function"
  | "no-root";

interface ValidationError {
  code: ValidationErrorCode;
  component: string;
  path: string;
  message: string;
  statementId?: string;
  /** "warning" when the program still renders, e.g. a no-root fallback entry. */
  severity?: "warning";
}

interface ParseResult {
  root: ElementNode | null;
  /** Set when the entry picks a component at runtime (`root = $t ? A : B`). */
  rootExpression?: ASTNode;
  meta: {
    incomplete: boolean;
    /** References used but not yet defined (dropped as null in output). */
    unresolved: string[];
    /** Value statements defined but not reachable from the entry. Excludes $state, Query, and Mutation. Empty when there is no entry. */
    orphaned: string[];
    statementCount: number;
    /**
     * Validation errors:
     * - "missing-required" — required prop not provided
     * - "null-required" — required prop explicitly null
     * - "unknown-component" — component not in library schema
     * - "inline-reserved" — Query/Mutation used inline instead of top-level
     * - "excess-args" — more positional args than schema params (extras dropped, component still renders)
     */
    errors: ValidationError[];
  };
  /** Extracted Query() statements with tool name, args AST, defaults AST */
  queryStatements: QueryStatementInfo[];
  /** Extracted Mutation() statements with tool name, args AST */
  mutationStatements: MutationStatementInfo[];
  /** Declared $variables with their default values */
  stateDeclarations: Record<string, unknown>;
}
```
